def get_torquantis_rfqs():
    """Fetch active RFQs from Torquantis"""
    response = requests.get(f"{TORQUANTIS_API_URL}/rfqs")
    return response.json()

def create_frantic_bounty(rfq_details):
    """Create a bounty on Frantic for a Torquantis RFQ"""
    bounty_data = {
        "title": f"Torquantis RFQ: {rfq_details['id']}",
        "description": f"Complete this Torquantis RFQ: {json.dumps(rfq_details)}",
        "amount": rfq_details['amount'],
        "currency": "USDC",
        "deadline": rfq_details['deadline'],
        "tags": ["torquantis", "rfq", rfq_details['work_type']]
    }

    headers = {"Authorization": f"Bearer {FRANTIC_API_KEY}"}
    response = requests.post(f"{FRANTIC_API_URL}/bounties", json=bounty_data, headers=headers)
    return response.json()

def submit_torquantis_quote(rfq_id, quote_amount):
    """Submit a quote to a Torquantis RFQ"""
    quote_data = {
        "rfq_id": rfq_id,
        "amount": quote_amount,
        "currency": "USDC"
    }
    response = requests.post(f"{TORQUANTIS_API_URL}/quotes", json=quote_data)
    return response.json()
